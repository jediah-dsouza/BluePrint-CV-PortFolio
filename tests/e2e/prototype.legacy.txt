test('hover and focus lift a region and illuminate the label', async ({ page }) => {
  await page.goto('/');
  const about = page.locator('.region-about');
  const initial = await about.evaluate((element) => getComputedStyle(element).transform);
  await about.hover();
  await page.waitForTimeout(250);
  const hovered = await about.evaluate((element) => getComputedStyle(element).transform);
  expect(hovered).not.toBe(initial);
  await expect(about.locator('.region-label')).toHaveCSS('color', 'rgb(216, 250, 255)');
  await page.mouse.move(0, 0); await about.focus();
  await expect(about.locator('.region-label')).toHaveCSS('color', 'rgb(216, 250, 255)');
});

for (const width of [390, 1440]) {
  test(`reduced motion holds plan and region transforms fixed at ${width}px`, async ({ page }) => {
    await page.setViewportSize({ width, height: 900 });
    await page.emulateMedia({ reducedMotion: 'reduce' }); await page.goto('/');
    await page.waitForSelector('html.js');
    const plan = page.locator('.plan');
    const baseline = await plan.evaluate(el => getComputedStyle(el).transform);
    for (const id of ['about', 'experience', 'education', 'skills', 'projects', 'contact']) {
      const region = page.locator(`.region-${id}`);
      const regionBaseline = await region.evaluate(el => getComputedStyle(el).transform);
      await region.focus();
      expect(await region.evaluate(el => getComputedStyle(el).transform)).toBe(regionBaseline);
      await page.keyboard.press('Enter');
      await expect(region).toHaveAttribute('aria-pressed', 'true');
      expect(await plan.evaluate(el => getComputedStyle(el).transform)).toBe(baseline);
      expect(await region.evaluate(el => getComputedStyle(el).transform)).toBe(regionBaseline);
      await expect(page.locator('.sheet')).toHaveCSS('animation-name', 'none');
      await expect(page.locator('.sheet')).toHaveCSS('transform', 'none');
    }
    await page.keyboard.press('Escape');
    expect(await plan.evaluate(el => getComputedStyle(el).transform)).toBe(baseline);
  });
}

for (const width of [768, 1024, 1440, 1920]) {
  test(`plan and selected regions remain framed at ${width}px`, async ({ page }) => {
    await page.setViewportSize({ width, height: 900 }); await page.goto('/');
    await page.waitForSelector('html.js');
    for (const id of ['overview', 'about', 'experience', 'education', 'skills', 'projects', 'contact']) {
      if (id !== 'overview') {
        await page.locator(`.region-${id}`).evaluate((el: HTMLElement) => el.click());
        await page.waitForTimeout(550);
      }
      await page.locator('.blueprint-frame').scrollIntoViewIfNeeded();
      const margins = await page.evaluate((id) => {
        const frame = document.querySelector('.blueprint-frame')!.getBoundingClientRect();
        const plan = document.querySelector('.plan')!.getBoundingClientRect();
        const item = id === 'overview' ? plan : document.querySelector(`.region-${id} .region-label`)!.getBoundingClientRect();
        const insets = (box: DOMRect) => [box.left - frame.left, frame.right - box.right, box.top - frame.top, frame.bottom - box.bottom];
        return { plan: insets(plan), label: insets(item) };
      }, id);
      expect(Math.min(...margins.plan), `${id} plan`).toBeGreaterThanOrEqual(20);
      expect(Math.min(...margins.label), `${id} label`).toBeGreaterThanOrEqual(20);
      const coveredLabels = await page.locator('.region-label').evaluateAll(labels => labels.flatMap(label => {
        const rect = label.getBoundingClientRect();
        const hit = document.elementFromPoint(rect.left + rect.width / 2, rect.top + rect.height / 2);
        return hit?.closest('.region') === label.closest('.region') ? [] : [label.textContent];
      }));
      expect(coveredLabels, `${id}: neighboring labels must remain uncovered`).toEqual([]);
    }
  });
}

for (const width of [360, 390]) test(`mobile touch switches and closes each section with adequate targets at ${width}px`, async ({ browser }) => {
  const context = await browser.newContext({ viewport: { width, height: 844 }, hasTouch: true, isMobile: true });
  const page = await context.newPage(); await page.goto('/');

  const close = page.locator('.sheet').getByRole('button', { name: /Close/ });
  const rect = await close.boundingBox(); expect(rect!.width).toBeGreaterThanOrEqual(44); expect(rect!.height).toBeGreaterThanOrEqual(44);
  await close.tap(); await expect(page.locator('.sheet')).toHaveCount(0);
  await expect(page.locator('.mobile-controls').getByRole('button', { name: /Contact/ })).toBeFocused();
  const menu = page.getByRole('button', { name: 'Menu' });
  const menuBox = await menu.boundingBox(); expect(menuBox!.width).toBeGreaterThanOrEqual(44); expect(menuBox!.height).toBeGreaterThanOrEqual(44);
  await menu.tap();
  for (const link of await page.locator('#header-nav a').all()) {
    const box = await link.boundingBox(); expect(box!.width).toBeGreaterThanOrEqual(44); expect(box!.height).toBeGreaterThanOrEqual(44);
  }
  await page.getByRole('link', { name: 'About', exact: true }).tap();
  await expect(page.locator('#sheet-title')).toHaveText('About');
  await expect(menu).toHaveAttribute('aria-expanded', 'false');
  await context.close();
});

for (const state of ['overview', 'sheet', 'mobile-menu', 'mobile-sheet', 'reduced-motion', 'fallback'] as const) {
  test(`automated accessibility check: ${state}`, async ({ page }) => {
    if (state === 'mobile-menu' || state === 'mobile-sheet') await page.setViewportSize({ width: 390, height: 844 });
    if (state === 'reduced-motion') await page.emulateMedia({ reducedMotion: 'reduce' });
    await page.goto('/');
    if (state === 'sheet' || state === 'reduced-motion') await page.locator('.region-about').click();
    if (state === 'mobile-sheet') await page.locator('.mobile-controls').getByRole('button', { name: /About/ }).click();
    if (state === 'mobile-menu') await page.getByRole('button', { name: 'Menu' }).click();
    if (state === 'fallback') await page.getByRole('button', { name: 'Use simplified plan' }).click();
    if (['sheet', 'mobile-sheet', 'reduced-motion'].includes(state)) {
      await page.locator('.sheet').evaluate(async el => { await Promise.all(el.getAnimations().map(animation => animation.finished)); });
    }
    await page.evaluate(axe.source);
    const violations = await page.evaluate(async () => {
      const result = await (window as unknown as { axe: typeof axe }).axe.run(document, { runOnly: { type: 'tag', values: ['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa'] } });
      return result.violations.map(v => ({ id: v.id, impact: v.impact, nodes: v.nodes.map(n => n.target) }));
    });
    expect(violations).toEqual([]);
  });
}
